# Пример формата `/proc/self/maps` (синтетический)
#
# Оригинальная библиотека ищет первую строку, где встречаются и ".apk",
# и "/data/app/", и берёт её пятое поле — номер inode. Больше ничего из
# файла не используется.
#
# Значение всё равно непроверяемо: оно лежит внутри зашифрованного токена,
# и сервер берёт его оттуда, а не с устройства. Поэтому публиковать реальный
# дамп не нужно — достаточно такого примера (или константы DEFAULT_INODE).
#
# start-end              perms offset   dev   inode   path
7f0000000000-7f0000100000 r--p 00000000 fd:00 656421  /data/app/~~AAAA==/com.example.app-XXXX==/base.apk
7f0000100000-7f0000200000 r-xp 00001000 fd:00 656421  /data/app/~~AAAA==/com.example.app-XXXX==/base.apk
7f0000200000-7f0000300000 r--p 00000000 fd:00 123456  /system/lib64/libc.so
